let userName = "Elzero";

console.log(userName.charAt(3));        // e
console.log(userName[3]);               // e
console.log(userName.slice(3, 4));      // e
console.log(userName.substring(3, 4));  // e
console.log(userName.at(3));            // e
console.log(userName[3].repeat(3));     // eee
