"02" — Six Ways to Get "6"

let num = 3;

// Solution One
console.log(num * (true + true)); // 6

// Solution Two
console.log(num + num); // 6

// Solution Three
console.log(num ** (true + true) - num); // 6

// Solution Four
console.log(num * (true + true + true) - num); // 6

// Solution Five
console.log(num / (true + true) + num); // 6

// Solution Six
console.log(num * num / num + num); // 6

Concepts

"+" Addition
"*" Multiplication
"/" Division
"**" Exponentiation
Boolean values
Operator precedence

"03" — Type Conversion

let num = "10";

// Solution One
console.log(num * (true + true)); // 20

// Solution Two
console.log(+num + +num); // 20

// Solution Three
console.log((true + true + true + true) * num / (true + true)); // 20

// Solution Four
console.log(+num * +num / +num + +num); // 20

Unary Plus "+"
String to Number Conversion
Arithmetic Operators

"04" — Increment & Decrement

let points = 10;

points++;
++points;
points++;

console.log(points); // 13

points--;
--points;
points--;
--points;
points--;

console.log(points); // 8

Concepts

"++" Increment
"--" Decrement
Prefix operators
Postfix operators
Variable updates
